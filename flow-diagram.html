Implementation (matcher.go)

func Match(actual, expected string) bool { a := strings.Fields(Normalize(actual)) e := strings.Fields(Normalize(expected)) if len(a) != len(e) { return false } for i := range e { if !tokenEqual(a[i], e[i]) { return false } } return true } func tokenEqual(a, e string) bool { if a == e { return true } af, aErr := strconv.ParseFloat(a, 64) ef, eErr := strconv.ParseFloat(e, 64) if aErr != nil || eErr != nil { return false } return math.Abs(af-ef)/math.Max(math.Abs(af), math.Abs(ef)) <= 1e-6 }
